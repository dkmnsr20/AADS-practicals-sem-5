1. *Randomized QuickSort*


#include <iostream>
#include <cstdlib>
#include <ctime>
using namespace std;

int comparisons = 0;

// Pick random pivot, then partition
int partition(int arr[], int low, int high) {
    swap(arr[low + rand() % (high - low + 1)], arr[high]);
    int i = low;
    for (int j = low; j < high; j++, comparisons++)
        if (arr[j] <= arr[high]) swap(arr[i++], arr[j]);
    swap(arr[i], arr[high]);
    return i;
}

// Randomized Quick Sort
void quickSort(int arr[], int low, int high) {
    if (low < high) {
        int p = partition(arr, low, high);
        quickSort(arr, low, p - 1);
        quickSort(arr, p + 1, high);
    }
}


int main() {
    srand(time(0));
    int arr[] = {34, 7, 23, 32, 5, 62, 32, 1};
    int n = 8;

    quickSort(arr, 0, n - 1);

    cout << "Sorted array: ";
    for (int i = 0; i < n; i++)
        cout << arr[i] << " ";

    cout << "\nNumber of comparisons: " << comparisons << endl;
    return 0;
}

2. *Randomized Select*

#include <iostream>
#include <cstdlib>
#include <ctime>
#include <algorithm>
using namespace std;

int a[100];

int kth(int l, int r, int k){
    // 1. One element left
    if(l == r) return a[l];

    // 2. Pick random pivot, move it to the end
    swap(a[l + rand() % (r - l + 1)], a[r]);

    // 3. Partition
    int i = l;

    for(int j = l; j < r; j++) {
        if(a[j] < a[r]) {
            swap(a[i], a[j]);
            i++;
        }
    }
    swap(a[i], a[r]);


    // 4. Found, go left, or go right
    if(i == k) return a[i];
    if(i > k) return kth(l, i - 1, k);
    return kth(i + 1, r, k);
}

int main(){
    int n, k;
    cout << "Enter size: "; cin >> n;
    cout << "Enter elements: ";
    for(int i = 0; i < n; i++) cin >> a[i];
    cout << "Enter k: "; cin >> k;
    srand(time(0));
    cout << "Kth smallest: " << kth(0, n - 1, k - 1);
}

3. *Kruskal algo*


#include <iostream>
using namespace std;

int main() {
    // each edge = {weight, u, v}
    int e[7][3] = {{2,0,1}, {6,0,3}, {3,1,2}, {8,1,3}, {5,1,4}, {7,2,4}, {9,3,4}};
    int parent[5] = {0, 1, 2, 3, 4};
    int cost = 0;

    // sort edges by weight
    for (int i = 0; i < 7; i++)
        for (int j = 0; j < 6; j++)
            if (e[j][0] > e[j+1][0])
                swap(e[j], e[j+1]);

    for (int i = 0; i < 7; i++) {
        int a = e[i][1], b = e[i][2];
        while (parent[a] != a) a = parent[a];
        while (parent[b] != b) b = parent[b];

        if (a != b) {   // no cycle, take the edge
            cout << e[i][1] << " - " << e[i][2] << " : " << e[i][0] << endl;
            cost += e[i][0];
            parent[a] = b;
        }
    }
    cout << "Total cost: " << cost << endl;
}

4. *Bellman Ford algo*


#include <iostream>
using namespace std;

int main() {
    int V, E, src, e[20][3], dist[20];

    cout << "Enter number of vertices and edges: ";
    cin >> V >> E;
    cout << "Enter each edge as u v weight:" << endl;
    for (int i = 0; i < E; i++)
        cin >> e[i][0] >> e[i][1] >> e[i][2];
    cout << "Enter source: ";
    cin >> src;

    for (int i = 0; i < V; i++)
        dist[i] = 99999;
    dist[src] = 0;

    // rounds 1 to V-1 relax the edges, round V checks for negative cycle
    for (int i = 1; i <= V; i++)
        for (int j = 0; j < E; j++) {
            int u = e[j][0], v = e[j][1], w = e[j][2];
            if (dist[u] != 99999 && dist[u] + w < dist[v]) {
                if (i == V) {
                    cout << "Negative cycle found" << endl;
                    return 0;
                }
                dist[v] = dist[u] + w;
            }
        }

    for (int i = 0; i < V; i++)
        cout << "Vertex " << i << " : " << dist[i] << endl;
}

5. *Suffix Trie*


#include <iostream>
using namespace std;

int trie[1000][128], total = 1;   // node 0 is the root, 0 means no edge

// insert every suffix of s into the trie, one by one
void bruteForceSuffixTrie(string s) {
    for (int i = 0; i < s.length(); i++) {
        cout << "Inserting: " << s.substr(i) << endl;
        int cur = 0;
        for (int j = i; j < s.length(); j++) {
            if (trie[cur][s[j]] == 0)
                trie[cur][s[j]] = total++;
            cur = trie[cur][s[j]];
        }
    }
}

int main() {
    string s, p;
    cout << "Enter text: ";
    cin >> s;
    bruteForceSuffixTrie(s + "$");

    cout << "Enter pattern: ";
    cin >> p;
    int cur = 0;
    for (int i = 0; i < p.length(); i++) {
        if (trie[cur][p[i]] == 0) {
            cout << "Not found" << endl;
            return 0;
        }
        cur = trie[cur][p[i]];
    }
    cout << "Found" << endl;
}

6. *KMP algo*


#include <iostream>
using namespace std;

int main() {
    string t, p;
    int lps[100] = {0};
    cout << "Enter text: ";
    cin >> t;
    cout << "Enter pattern: ";
    cin >> p;
    int n = t.length(), m = p.length();

    // build LPS
    int len = 0;
    for (int i = 1; i < m; i++) {
        while (len > 0 && p[i] != p[len])
            len = lps[len - 1];
        if (p[i] == p[len])
            len++;
        lps[i] = len;
    }

    // search
    int j = 0;
    for (int i = 0; i < n; i++) {
        while (j > 0 && t[i] != p[j])
            j = lps[j - 1];
        if (t[i] == p[j])
            j++;
        if (j == m) {
            cout << "Pattern found at index " << i - m + 1 << endl;
            j = lps[j - 1];
        }
    }
}

7. *B Tree*


#include <iostream>
using namespace std;

int key[200][3], ch[200][4], cnt[200], tot = 1;   // max 2 keys per node

void put(int x, int i, int k, int r) {   // put k at position i of node x
    for (int j = cnt[x]; j > i; j--) {
        key[x][j] = key[x][j - 1];
        ch[x][j + 1] = ch[x][j];
    }
    key[x][i] = k;
    ch[x][i + 1] = r;
    cnt[x]++;
}

void insert(int x, int k) {
    int i = 0;
    while (i < cnt[x] && key[x][i] < k) i++;
    if (!ch[x][0]) { put(x, i, k, 0); return; }   // leaf
    int c = ch[x][i];
    insert(c, k);
    if (cnt[c] == 3) {                     // split c
        int z = ++tot;
        key[z][0] = key[c][2];
        ch[z][0] = ch[c][2];
        ch[z][1] = ch[c][3];
        cnt[z] = cnt[c] = 1;
        put(x, i, key[c][1], z);           // middle key goes up
    }
}

void print(int x, int lv) {
    cout << "Level " << lv << ": ";
    for (int i = 0; i < cnt[x]; i++) cout << key[x][i] << " ";
    cout << endl;
    if (ch[x][0])
        for (int i = 0; i <= cnt[x]; i++) print(ch[x][i], lv + 1);
}

int main() {
    int n, k, root = 1;
    cin >> n;
    while (n--) {
        cin >> k;
        int s = ++tot;                     // empty node above root
        ch[s][0] = root;
        insert(s, k);
        if (cnt[s]) root = s;              // root split, s is new root
    }
    print(root, 0);
}
